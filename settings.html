// worker.js /** * Taskitator Cloudflare Worker Bridge * * - Multi-tenant cloud synchronization via partitioned KV storage. * - Zero-knowledge client-side token verification with username collision protection. * - Test connection ping endpoint for Settings. * - MacroDroid device-lock evaluation engine: checks task completion and active break windows. * * Required Binding: * - KV Namespace: TASKITATOR_KV */ const CORS_HEADERS = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, Authorization, X-Taskitator-User', 'Access-Control-Max-Age': '86400', }; // Local timezone offset in hours (UTC+3 for Riyadh) const TIMEZONE_OFFSET_HOURS = 3; export default { async fetch(request, env) { // 1. Preflight CORS handling if (request.method === 'OPTIONS') { return new Response(null, { status: 204, headers: CORS_HEADERS }); } try { // 2. Authentication check const authHeader = request.headers.get('Authorization') || ''; if (!authHeader || !authHeader.toLowerCase().startsWith('bearer ')) { return new Response(JSON.stringify({ error: 'Unauthorized: Missing or malformed Bearer token.' }), { status: 401, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } const secretKey = authHeader.replace(/^Bearer\s+/i, '').trim(); if (!secretKey) { return new Response(JSON.stringify({ error: 'Unauthorized: Empty token.' }), { status: 401, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } // 3. Username validation & collision protection const rawUser = request.headers.get('X-Taskitator-User') || ''; const cleanUser = rawUser.trim().toLowerCase(); if (cleanUser) { const regKey = `user_reg:${cleanUser}`; const registration = await env.TASKITATOR_KV.get(regKey, { type: 'json' }); if (!registration) { // First time seeing this username: register it to this secret token hash await env.TASKITATOR_KV.put(regKey, JSON.stringify({ token_hash: secretKey, created_at: new Date().toISOString() })); } else if (registration.token_hash !== secretKey) { // Collision: Username exists, but token hash does not match return new Response(JSON.stringify({ error: 'Unauthorized: Incorrect credentials or username already claimed.' }), { status: 401, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } } const url = new URL(request.url); const kvKey = `user_data:${secretKey}`; // ========================================================================= // GET Requests // ========================================================================= if (request.method === 'GET') { const action = url.searchParams.get('action'); // --- Route A: Status / Health Ping & MacroDroid Evaluator --- if (action === 'status') { const record = await env.TASKITATOR_KV.get(kvKey, { type: 'json' }); if (!record) { return buildStatusResponse({ status: 'ok', device_unlocked: true, finished: true, in_break: false, message: 'No tasks configured; unlocked by default.' }, url); } const tasks = Array.isArray(record.tasks) ? record.tasks : []; const breaks = Array.isArray(record.today_breaks) ? record.today_breaks : []; // Adjust UTC to target local timezone const nowUTC = new Date(); const localNow = new Date(nowUTC.getTime() + (TIMEZONE_OFFSET_HOURS * 60 * 60 * 1000)); const todayStr = localNow.toISOString().split('T')[0]; const currentMinutes = localNow.getUTCHours() * 60 + localNow.getUTCMinutes(); // 1. Evaluate Task Completion const hasUnfinishedToday = tasks.some(t => { if (t.status !== 'active') return false; const due = String(t.due_date || 'today').trim().toLowerCase(); return due === 'today' || due === todayStr; }); const isFinished = !hasUnfinishedToday; // 2. Evaluate Active Break Interval const isInBreak = breaks.some(b => { if (!b || !b.start || !b.end) return false; if (b.date && b.date !== todayStr) return false; if (b.start.includes('T')) { const startTime = new Date(b.start).getTime(); const endTime = new Date(b.end).getTime(); const cur = nowUTC.getTime(); return cur >= startTime && cur <= endTime; } const [startH, startM] = b.start.split(':').map(Number); const [endH, endM] = b.end.split(':').map(Number); const startMins = startH * 60 + startM; const endMins = endH * 60 + endM; if (startMins <= endMins) { return currentMinutes >= startMins && currentMinutes <= endMins; } else { // Overnight break return currentMinutes >= startMins || currentMinutes <= endMins; } }); const isDeviceUnlocked = isFinished || isInBreak; return buildStatusResponse({ status: 'ok', device_unlocked: isDeviceUnlocked, finished: isFinished, in_break: isInBreak }, url); } // --- Route B: Standard Sync Pull --- const record = await env.TASKITATOR_KV.get(kvKey, { type: 'json' }); if (!record) { return new Response(JSON.stringify({ empty: true, message: 'No cloud record found.' }), { status: 200, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } return new Response(JSON.stringify(record), { status: 200, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } // ========================================================================= // POST Requests (Sync Push) // ========================================================================= if (request.method === 'POST') { const incoming = await request.json().catch(() => null); if (!incoming || !Array.isArray(incoming.tasks)) { return new Response(JSON.stringify({ error: 'Invalid payload: tasks array required.' }), { status: 400, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } const existing = await env.TASKITATOR_KV.get(kvKey, { type: 'json' }); // Conflict detection: prevent stale overwrites unless forced if (existing && !incoming.force && existing.updated_at && incoming.updated_at) { const remoteTime = new Date(existing.updated_at).getTime(); const localTime = new Date(incoming.updated_at).getTime(); if (remoteTime > localTime) { return new Response(JSON.stringify({ error: 'Conflict: Remote data is newer than local state.', conflict: true, remote_updated_at: existing.updated_at }), { status: 409, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } } const newVersion = existing && existing.version ? existing.version + 1 : 1; const recordToStore = { ...incoming, version: newVersion, saved_at: new Date().toISOString() }; await env.TASKITATOR_KV.put(kvKey, JSON.stringify(recordToStore)); return new Response(JSON.stringify({ success: true, version: newVersion, updated_at: recordToStore.updated_at }), { status: 200, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } return new Response(JSON.stringify({ error: 'Method Not Allowed' }), { status: 405, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } catch (err) { return new Response(JSON.stringify({ error: 'Internal Worker Exception: ' + err.message }), { status: 500, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); } } }; /** * Format helper for MacroDroid and Settings test responses */ function buildStatusResponse(state, url) { const format = url.searchParams.get('format'); if (format === 'text') { return new Response(`device_unlocked = ${state.device_unlocked}`, { status: 200, headers: { ...CORS_HEADERS, 'Content-Type': 'text/plain; charset=utf-8' } }); } return new Response(JSON.stringify(state), { status: 200, headers: { ...CORS_HEADERS, 'Content-Type': 'application/json' } }); }
